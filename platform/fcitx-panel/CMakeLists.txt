cmake_minimum_required(VERSION 3.21)
project(nextkde-fcitx-panel LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Fcitx5Core 5.0.24 REQUIRED)
find_package(Fcitx5ModuleDBus REQUIRED)
add_library(nextkde-inputpanel MODULE NextKdePanel.cpp)
set_target_properties(nextkde-inputpanel PROPERTIES AUTOMOC OFF)
target_compile_features(nextkde-inputpanel PRIVATE cxx_std_20)
target_link_libraries(nextkde-inputpanel PRIVATE Fcitx5::Core Fcitx5::Module::DBus)
set(NEXTKDE_PANEL_LIBRARY "${CMAKE_INSTALL_FULL_LIBDIR}/fcitx5/libnextkde-inputpanel")
configure_file(nextkde-inputpanel.conf nextkde-inputpanel.conf @ONLY)
install(TARGETS nextkde-inputpanel LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/fcitx5)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/nextkde-inputpanel.conf
    DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/addon)
