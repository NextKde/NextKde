cmake_minimum_required(VERSION 3.21)
project(KosAppDependencies LANGUAGES C CXX)
if(KOS_WITH_LISTENFREE)
    find_package(Qt6 6.10 QUIET COMPONENTS Core Widgets Network DBus)
else()
    find_package(Qt6 6.5 QUIET COMPONENTS Core Network DBus)
endif()
foreach(package IN ITEMS Qt6LinguistTools KF6WindowSystem)
    find_package(${package} QUIET)
    if(NOT ${package}_FOUND)
        message(STATUS "KOS_MISSING:${package}")
    endif()
endforeach()
if(NOT KOS_SDK_ONLY)
    find_package(KF6CalendarCore 6.0 QUIET)
    if(NOT KF6CalendarCore_FOUND)
        message(STATUS "KOS_MISSING:KF6CalendarCore")
    endif()
endif()
if(KOS_BUILD_TAGLIB)
    find_path(KOS_UTF8CPP_INCLUDE NAMES utf8.h PATH_SUFFIXES utf8cpp)
    if(NOT KOS_UTF8CPP_INCLUDE)
        message(STATUS "KOS_MISSING:Utf8Cpp")
    endif()
endif()
# Ubuntu splits QML runtime modules from development headers.
if(TARGET Qt6::qtpaths)
    get_target_property(qtpaths Qt6::qtpaths IMPORTED_LOCATION)
    if(NOT qtpaths)
        get_target_property(configurations Qt6::qtpaths IMPORTED_CONFIGURATIONS)
        foreach(configuration IN LISTS configurations)
            get_target_property(qtpaths Qt6::qtpaths IMPORTED_LOCATION_${configuration})
            if(qtpaths)
                break()
            endif()
        endforeach()
    endif()
    execute_process(COMMAND "${qtpaths}" --query QT_INSTALL_QML OUTPUT_VARIABLE qml_root OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE qml_query)
    if(NOT qml_query EQUAL 0)
        message(FATAL_ERROR "Cannot query QML runtime path from the selected Qt toolchain")
    endif()
    execute_process(COMMAND "${qtpaths}" --query QT_INSTALL_PLUGINS OUTPUT_VARIABLE plugin_root OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT EXISTS "${plugin_root}/sqldrivers/libqsqlite.so")
        message(STATUS "KOS_MISSING:SQLiteDriver")
    endif()
    set(qml_modules QtCore QtQml QtQml/Models QtQml/WorkerScript QtQuick/Effects QtQuick/Shapes QtQuick QtQuick/Window QtQuick/Layouts QtQuick/Controls QtQuick/Templates QtQuick/Dialogs)
    if(LISTENFREE_BUILD_VIDEO)
        list(APPEND qml_modules QtMultimedia)
    endif()
    foreach(module IN LISTS qml_modules)
        if(NOT EXISTS "${qml_root}/${module}/qmldir")
            string(REPLACE "/" "_" label "${module}")
            message(STATUS "KOS_MISSING:Qml_${label}")
        endif()
    endforeach()
endif()
